--// Asset helper
local function loadAsset(url, filename)
    local name = filename or "hook_bg.png"
    local getCustomAsset = getcustomasset or getsynasset or (syn and syn.get_custom_asset)
    if not getCustomAsset then return "" end

    if isfile and isfile(name) then
        local ok, result = pcall(getCustomAsset, name)
        if ok and result and result ~= "" then
            return result
        end
    end

    if writefile then
        pcall(function()
            local request_ = syn and syn.request
                or http and http.request
                or http_request
                or fluxus and fluxus.request
                or request
            local body = nil

            if request_ then
                local response = request_({ Url = url, Method = "GET" })
                local body2 = response and (response.Body or response.body)
                if body2 then
                    body = response.Body or response.body
                end
            end

            if not body and game.HttpGet then
                body = game:HttpGet(url)
            end

            if body and #body > 0 then
                writefile(name, body)
            end
        end)

        if isfile and isfile(name) then
            local ok, result = pcall(getCustomAsset, name)
            if ok and result and result ~= "" then
                return result
            end
        end
    end
    return ""
end

--// Set hidden property helper
local function setHidden(obj, prop, value)
    if not obj then return false end
    local setter = sethiddenproperty or set_hidden_property or sethiddenprop or set_hidden_prop
    if setter then
        if pcall(setter, obj, prop, value) then return true end
    end
    return pcall(function()
        obj[prop] = value
    end)
end

--// Physics boost
local function boostPhysics()
    pcall(function()
        if setsimulationradius then
            setsimulationradius(math.huge, math.huge)
        end

        setHidden(LocalPlayer, "MaximumSimulationRadius", math.huge)
        setHidden(LocalPlayer, "SimulationRadius", math.huge)
        setHidden(workspace, "InterpolationThrottling", Enum.InterpolationThrottlingMode.Disabled)
        setHidden(workspace, "FallenPartsDestroyHeight", -500000)

        pcall(function()
            workspace.FallenPartsDestroyHeight = -500000
        end)

        if settings().Physics then
            pcall(function()
                settings().Physics.PhysicsEnvironmentalThrottle = Enum.EnviromentalPhysicsThrottle.Disabled
                settings().Physics.AllowSleep = false
            end)
        end

        if settings().Network then
            pcall(function()
                settings().Network.InterpolationThrottling = Enum.InterpolationThrottlingMode.Disabled
            end)
        end

        if NetworkClient and NetworkClient.SetOutgoingKBPSLimit then
            NetworkClient:SetOutgoingKBPSLimit(math.huge)
        end
    end)
end

--// Get root part
local function getRoot(character)
    if not character then return nil end
    return character:FindFirstChild("HumanoidRootPart")
        or character:FindFirstChild("Torso")
        or character:FindFirstChild("UpperTorso")
        or character.PrimaryPart
end

--// Find nearest player head
local function findNearestHead(fromPos)
    if not fromPos then return nil end

    local closest = math.huge
    local bestPos = nil

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local root = getRoot(player.Character)
            local humanoid = player.Character:FindFirstChildOfClass("Humanoid")

            if root and (not humanoid or humanoid.Health > 0) then
                local dist = (fromPos.Position - root.Position).Magnitude
                if dist < closest then
                    local head = player.Character:FindFirstChild("Head")
                    if head then
                        bestPos = head.Position + Vector3.new(0, 2, 0)
                        closest = dist
                    else
                        bestPos = root.Position + Vector3.new(0, 5, 0)
                        closest = dist
                    end
                end
            end
        end
    end
    return bestPos
end

--// Get/create anchor part
local function getAnchorPart(cf)
    if not anchorPart or not anchorPart.Parent or not anchorPart:IsDescendantOf(workspace) then
        pcall(function()
            if anchorPart then anchorPart:Destroy() end
        end)

        anchorPart = Instance.new("Part")
        anchorPart.Name = "HookAntiAntiAnchor"
        anchorPart.Size = Vector3.new(2, 2, 1)
        anchorPart.Anchored = true
        anchorPart.CanCollide = false
        anchorPart.CanTouch = false
        anchorPart.CanQuery = false
        anchorPart.Transparency = 1
        anchorPart.Massless = true
        if cf then anchorPart.CFrame = cf end
        anchorPart.Parent = workspace
    end
    return anchorPart
end

    --// Main fling logic
    local function flingStep()
        if not config.enabled then return end
        local character = LocalPlayer.Character
        if not character then return end

        local root = getRoot(character)
        if not root then return end

        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if humanoid then
            pcall(function()
                humanoid:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
                humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
                humanoid:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
            end)
        end

        boostPhysics()

        --// Disable collision on other players
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                for _, descendant in ipairs(player.Character:GetDescendants()) do
                    if descendant:IsA("BasePart") and descendant.CanCollide then
                        descendant.CanCollide = false
                    end
                end
            end
        end

        local anchor = getAnchorPart()
        if not anchor then return end

        local angleCFrame = CFrame.Angles(
            math.rad(math.random(0, 360)),
            math.rad(math.random(0, 360)),
            math.rad(math.random(0, 360))
        )

        pcall(function()
            if root.AssemblyLinearVelocity.Magnitude < 0.1 then
                root.AssemblyLinearVelocity = Vector3.new(0, 0.01, 0)
            end
        end)

        if config.version == "V1" then
            --// V1: Spiral fling
            if not position then position = root.Position end
            if not savedCFrame then savedCFrame = root.CFrame end

            if cancelTween then
                pcall(function() cancelTween:Cancel() end)
                cancelTween = nil
            end

            pcall(function()
                local look = root.CFrame.LookVector
                if look.Magnitude < 0.01 and savedCFrame then
                    look = savedCFrame.LookVector
                end
                local flatLook = Vector3.new(look.X, 0, look.Z)
                if flatLook.Magnitude < 0.01 then
                    flatLook = Vector3.new(0, 0, -1)
                end
                root.CFrame = CFrame.new(position, position + flatLook)
                root.AssemblyLinearVelocity = Vector3.new(
                    (math.random() - 0.5) * 0.04, 0.02,
                    (math.random() - 0.5) * 0.04
                )
                root.AssemblyAngularVelocity = Vector3.zero
            end)

            versionNum = (versionNum + 1.6) % 6.2831853071795862
            rotationAngle = rotationAngle + 1
            local phase = rotationAngle % 4
            local basePos = position
            local targetVec

            if phase == 0 then
                local angle = math.random() * math.pi * 2
                local radius = math.random(8, 14)
                targetVec = Vector3.new(
                    basePos.X + math.cos(angle) * radius,
                    basePos.Y + math.random(0, 1),
                    basePos.Z + math.sin(angle) * radius
                )
            elseif phase == 1 then
                local angle = math.random() * math.pi * 2
                local radius = math.random(4, 9)
                targetVec = Vector3.new(
                    basePos.X + math.cos(angle) * radius,
                    basePos.Y,
                    basePos.Z + math.sin(angle) * radius
                )
            elseif phase == 2 then
                local angle = math.random() * math.pi * 2
                local radius = math.random(3, 8)
                targetVec = Vector3.new(
                    basePos.X + math.cos(angle) * radius,
                    basePos.Y + math.random(18, 28),
                    basePos.Z + math.sin(angle) * radius
                )
            else
                targetVec = Vector3.new(
                    basePos.X + math.random(-12, 12),
                    basePos.Y + math.random(5, 20),
                    basePos.Z + math.random(-12, 12)
                )
            end

            local finalVec
            if rotationAngle % 3 == 1 then
                finalVec = Vector3.new(
                    targetVec.X + math.random(-40, 40),
                    -500 - math.random(0, 20000),
                    targetVec.Z + math.random(-40, 40)
                )
            else
                finalVec = targetVec + Vector3.new(
                    (math.random() - 0.5) * 1.5,
                    (math.random() - 0.5) * 1.5,
                    (math.random() - 0.5) * 1.5
                )
            end

            anchor.CFrame = CFrame.new(finalVec) * angleCFrame
            anchor.AssemblyLinearVelocity = Vector3.new(
                math.random(-100, 100),
                math.random(-9999999, 9999999),
                math.random(-100, 9999999)
            )
            anchor.AssemblyAngularVelocity = Vector3.new(
                math.random(-9999999, 100),
                math.random(-9999999, 100),
                math.random(-9999999, 9999999)
            )

        elseif config.version == "V5" then
            --// V5: Bobbing fling
            local pos = root.Position
            versionNum = (versionNum + 1.2) % 6.2831853071795862
            local bobVec = Vector3.new(
                pos.X,
                pos.Y + math.clamp((math.sin(versionNum * 4) + 1) * 22.5, 0, 45),
                pos.Z
            )
            anchor.CFrame = CFrame.new(bobVec) * angleCFrame
            anchor.AssemblyLinearVelocity = Vector3.new(0, math.cos(versionNum * 4) * 50, 0)
            anchor.AssemblyAngularVelocity = Vector3.new(0, 30, 0)

        elseif config.version == "V4" then
            --// V4: Circular orbit fling
            if not position then position = root.Position end
            local smallVel = Vector3.new(
                (math.random() - 0.5) * 0.04, 0.02,
                (math.random() - 0.5) * 0.04
            )

            pcall(function()
                if root and position then
                    root.CFrame = CFrame.new(position, position + root.CFrame.LookVector)
                    root.AssemblyLinearVelocity = smallVel
                end
            end)

            versionNum = (versionNum + 0.95) % 6.2831853071795862
            local radius = 14 + math.sin(versionNum * 3.5) * 6
            local xOff = math.cos(versionNum) * radius
            local zOff = math.sin(versionNum) * radius
            local yOff = math.sin(versionNum * 4) * 4.5 + math.cos(versionNum * 2.5) * 2.5
            local jitter = Vector3.new(
                (math.random() - 0.5) * 0.4,
                (math.random() - 0.5) * 0.4,
                (math.random() - 0.5) * 0.4
            )
            anchor.CFrame = CFrame.new(position + Vector3.new(xOff, yOff, zOff) + jitter) * angleCFrame
            anchor.AssemblyLinearVelocity = Vector3.new(
                math.random(-100, 100),
                -9999999,
                math.random(-9999999, 9999999)
            )
            anchor.AssemblyAngularVelocity = Vector3.new(
                math.random(-9999999, 100),
                math.random(-9999999, 9999999),
                math.random(-100, 100)
            )

        elseif config.version == "V3" then
            --// V3: Head-target fling
            local target = findNearestHead(root) or root.Position + Vector3.new(0, 4.5, 0)
            anchor.CFrame = CFrame.new(target + Vector3.new(
                (math.random() - 0.5) * 0.08,
                (math.random() - 0.5) * 0.08,
                (math.random() - 0.5) * 0.08
            )) * angleCFrame
            anchor.AssemblyLinearVelocity = Vector3.new(
                math.random(-9999999, 9999999),
                -100,
                math.random(-9999999, 100)
            )
            anchor.AssemblyAngularVelocity = Vector3.new(
                math.random(-9999999, 100),
                math.random(-9999999, 9999999),
                math.random(-100, 9999999)
            )

        elseif config.version == "V2" then
            --// V2: Fixed position fling
            local targetPos = config.targetPos or Vector3.new(-538.104, 7.471, 60.085)
            anchor.CFrame = CFrame.new(targetPos + Vector3.new(
                (math.random() - 0.5) * 30,
                (math.random() - 0.5) * 0.06,
                (math.random() - 0.5) * 0.06
            )) * angleCFrame
            anchor.AssemblyLinearVelocity = Vector3.new(
                math.random(-9999999, 9999999),
                -9999999,
                math.random(-9999999, 9999999)
            )
            anchor.AssemblyAngularVelocity = Vector3.new(
                math.random(-9999999, 9999999),
                math.random(-9999999, 100),
                math.random(-9999999, 9999999)
            )

        else
            --// V6: Random teleport fling
            anchor.CFrame = CFrame.new(
                root.Position.X + math.random(-100000, 100000),
                -500000 - math.random(10000, 100000),
                root.Position.Z + math.random(-100000, 100000)
            ) * angleCFrame
            anchor.AssemblyLinearVelocity = Vector3.new(
                math.random(-9999999, 9999999),
                -100,
                math.random(-9999999, 9999999)
            )
            anchor.AssemblyAngularVelocity = Vector3.new(
                math.random(-100, 100),
                math.random(-9999999, 9999999),
                math.random(-9999999, 9999999)
            )
        end

        setHidden(root, "PhysicsRepRootPart", root)
        setHidden(root, "PhysicsRepRootPart", anchor)
    end

    --// Cleanup
    local function cleanupConnections()
        for _, conn in ipairs(connections) do
            if conn then
                pcall(function() conn:Disconnect() end)
            end
        end
        table.clear(connections)
        position = nil
        savedCFrame = nil
        versionNum = 0
        rotationAngle = 0

        if cancelTween then
            pcall(function() cancelTween:Cancel() end)
            cancelTween = nil
        end

        if thread then
            pcall(function() task.cancel(thread) end)
            thread = nil
        end

        pcall(function()
            local root = getRoot(LocalPlayer.Character)
            if root then
                pcall(function()
                    root.AssemblyLinearVelocity = Vector3.zero
                    root.AssemblyAngularVelocity = Vector3.zero
                end)

                if anchorPart then
                    anchorPart.CFrame = root.CFrame
                    anchorPart.AssemblyLinearVelocity = Vector3.zero
                    anchorPart.AssemblyAngularVelocity = Vector3.zero
                end

                setHidden(root, "PhysicsRepRootPart", root)

                pcall(function()
                    root:ApplyImpulse(Vector3.new(0, 0.001, 0))
                end)
            end
        end)
    end

    --// Start/stop fling
    local function setFling(enabled)
        if not enabled then
            cleanupConnections()
            return
        end

        boostPhysics()
        local root = getRoot(LocalPlayer.Character)
        if root then
            position = root.Position
            savedCFrame = root.CFrame
        end

        versionNum = 0
        rotationAngle = 0
        local anchor = getAnchorPart(root and root.CFrame)

        if root and anchor then
            if root.CFrame then anchor.CFrame = root.CFrame end
            setHidden(root, "PhysicsRepRootPart", root)
            setHidden(root, "PhysicsRepRootPart", anchor)
        end

        if #connections == 0 then
            table.insert(connections, RunService.Stepped:Connect(flingStep))
            table.insert(connections, RunService.Heartbeat:Connect(flingStep))
            table.insert(connections, RunService.RenderStepped:Connect(flingStep))

            if pcall(function() return RunService.PreRender end) and RunService.PreRender then
                pcall(function()
                    table.insert(connections, RunService.PreRender:Connect(flingStep))
                end)
            end

            if pcall(function() return RunService.PreSimulation end) and RunService.PreSimulation then
                pcall(function()
                    table.insert(connections, RunService.PreSimulation:Connect(flingStep))
                end)
            end

            if pcall(function() return RunService.PostSimulation end) and RunService.PostSimulation then
                pcall(function()
                    table.insert(connections, RunService.PostSimulation:Connect(flingStep))
                end)
            end
        end

        if not thread then
            thread = task.spawn(function()
                while config.enabled do
                    flingStep()
                    task.wait()
                end
            end)
        end
    end

                    end
                end
            end
        end)
    end)

    --// Instance creation helper

        local instance = Instance.new(className)
        for k, v in pairs(props or {}) do
            if k ~= "Parent" then
                instance[k] = v
            end
        end
        for _, child in ipairs(children or {}) do
            child.Parent = instance
        end
        if props and props.Parent then
            instance.Parent = props.Parent
        end
        return instance
    end

--// Ripple effect
local function createRipple(parent, x, y, color)
    local ImageLabel = create("ImageLabel", {
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.fromOffset(x, y),
        Size = UDim2.fromOffset(0, 0),
        BackgroundTransparency = 1,
        Image = Assets.Circle,
        ImageColor3 = color or Color3.fromRGB(255, 70, 239),
        ImageTransparency = 0.4,
        ZIndex = 50,
        Parent = parent,
    })

    local maxSize = math.max(parent.AbsoluteSize.X, parent.AbsoluteSize.Y) * 2.2
    local tween = TweenService:Create(ImageLabel, TweenInfo.new(0.55, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {
        Size = UDim2.fromOffset(maxSize, maxSize),
        ImageTransparency = 1,
    })
    tween:Play()
    tween.Completed:Connect(function() ImageLabel:Destroy() end)
end

    if gethui then
        local ok, parent = pcall(gethui)
        if ok and parent then return parent end
    end

    local ok, parent = pcall(function() return game:GetService("CoreGui") end)
    if ok and parent then
        if syn and syn.protect_gui then
            pcall(syn.protect_gui, parent)
        end
        return parent
    end
    return LocalPlayer:WaitForChild("PlayerGui")
end

--// Keybind listener
do
    local function matchesKeybind(input, keyName)
        if not keyName or keyName == "" or keyName == "None" then return false end
        local isKeyboard = input.UserInputType == Enum.UserInputType.Keyboard
        local isGamepad = input.UserInputType.Name:find("Gamepad")
        if isKeyboard or isGamepad then
            return input.KeyCode.Name == keyName
        end
        return false
    end

    UserInputService.InputBegan:Connect(function(input, gameProcessed)
        if isSelectingKeybind then return end
        local notProcessed = not gameProcessed

        if notProcessed and matchesKeybind(input, config.keybind) then
            toggleUI()
        end

        if notProcessed and input.KeyCode == Enum.KeyCode.RightControl then
            config.guiVisible = not config.guiVisible
            Frame.Visible = config.guiVisible
        end
    end)
